class SettlementService(
  private val ledger: LedgerPort,
  private val events: EventBus
) {
  fun settle(cmd: SettleCommand): Single<Receipt> =
    ledger.debit(cmd.amount)
      .flatMap { events.publish(cmd) }
      .map { Receipt.ok(cmd.id) }
}
